-- ============================================================
-- 1. DATABASE
-- ============================================================

CREATE DATABASE testdb;
USE testdb;

-- ============================================================
-- 2. TABLE
-- ============================================================

CREATE TABLE users (
    id int INDEXED,
    name string NOT_NULL,
    age int,
    city string,
    email string INDEXED,
    bio string
);

-- ============================================================
-- 3. INSERT
-- ============================================================

INSERT INTO users (id, name, age, city, email, bio)
VALUE
    (1, "Alice", 25, "Moscow", "alice@test.com", "Developer"),
    (2, "Bob", 30, "Berlin", "bob@test.com", "Engineer"),
    (3, "Charlie", 20, "London", "charlie@test.com", "Student"),
    (4, "Diana", 35, "Paris", "diana@test.com", "Manager"),
    (5, "Eve", 28, "Moscow", "eve@test.com", "Designer");

-- Проверка полного SELECT
SELECT * FROM users;

-- ============================================================
-- 4. INSERT С ПРОПУЩЕННЫМИ NULLABLE ПОЛЯМИ
-- ============================================================

INSERT INTO users (id, name, email)
VALUE
    (6, "Frank", "frank@test.com");

SELECT * FROM users;

-- ============================================================
-- 5. SELECT КОНКРЕТНЫХ КОЛОНОК
-- ============================================================

SELECT id, name, age FROM users;

-- ============================================================
-- 6. SELECT С ALIAS
-- ============================================================

SELECT
    id AS user_id,
    name AS username,
    age AS user_age
FROM users;

-- ============================================================
-- 7. == 
-- ============================================================

SELECT * FROM users WHERE id == 3;

SELECT * FROM users WHERE name == "Alice";

SELECT * FROM users WHERE city == "Moscow";

-- ============================================================
-- 8. !=
-- ============================================================

SELECT * FROM users WHERE age != 30;

SELECT * FROM users WHERE city != "Moscow";

-- ============================================================
-- 9. <
-- ============================================================

SELECT * FROM users WHERE age < 30;

SELECT * FROM users WHERE name < "Diana";

-- ============================================================
-- 10. >
-- ============================================================

SELECT * FROM users WHERE age > 25;

SELECT * FROM users WHERE name > "Bob";

-- ============================================================
-- 11. <=
-- ============================================================

SELECT * FROM users WHERE age <= 28;

-- ============================================================
-- 12. >=
-- ============================================================

SELECT * FROM users WHERE age >= 30;

-- ============================================================
-- 13. BETWEEN
-- Интервал [20, 30)
-- ============================================================

SELECT * FROM users WHERE age BETWEEN 20 AND 30;

SELECT * FROM users WHERE id BETWEEN 2 AND 5;

-- ============================================================
-- 14. LIKE / REGEX
-- ============================================================

SELECT * FROM users WHERE name LIKE "^A.*";

SELECT * FROM users WHERE name LIKE ".*e$";

SELECT * FROM users WHERE city LIKE "^M.*";

SELECT * FROM users WHERE email LIKE ".*@test\\.com$";

-- ============================================================
-- 15. СРАВНЕНИЕ С NULL
-- ============================================================

SELECT * FROM users WHERE bio == NULL;

SELECT * FROM users WHERE bio != NULL;

-- ============================================================
-- 16. UPDATE
-- ============================================================

UPDATE users
SET age = 26
WHERE id == 1;

SELECT * FROM users WHERE id == 1;

-- ============================================================
-- 17. UPDATE НЕСКОЛЬКИХ КОЛОНОК
-- ============================================================

UPDATE users
SET age = 31, city = "Munich"
WHERE id == 2;

SELECT * FROM users WHERE id == 2;

-- ============================================================
-- 18. UPDATE ПО НЕИНДЕКСИРОВАННОМУ ПОЛЮ
-- ============================================================

UPDATE users
SET bio = "Senior Engineer"
WHERE city == "Munich";

SELECT * FROM users WHERE city == "Munich";

-- ============================================================
-- 19. DELETE
-- ============================================================

DELETE FROM users WHERE id == 6;

SELECT * FROM users;

-- ============================================================
-- 20. DELETE ПО ДИАПАЗОНУ
-- ============================================================

DELETE FROM users WHERE age BETWEEN 20 AND 25;

SELECT * FROM users;

-- ============================================================
-- 21. ПРОВЕРКА INDEXED ПОИСКА
-- ============================================================

SELECT * FROM users WHERE id == 2;

SELECT * FROM users WHERE id == 4;

SELECT * FROM users WHERE email == "alice@test.com";

SELECT * FROM users WHERE email == "diana@test.com";

-- ============================================================
-- 22. ПОЛНЫЙ DIAGNOSTIC SELECT
-- ============================================================

SELECT
    id AS identifier,
    name AS full_name,
    age AS years,
    city AS location,
    email AS contact,
    bio AS description
FROM users;

-- ============================================================
-- 23. DROP TABLE
-- ============================================================

DROP TABLE users;

-- ============================================================
-- 24. DROP DATABASE
-- ============================================================

DROP DATABASE testdb;
